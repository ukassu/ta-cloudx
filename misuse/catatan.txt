cd /home/ukass/cloudx-fp/misuse/boilerplate/data_generator

# a) Image GPU (DIPAKAI untuk dataset utama) -- torch CUDA 12.8, ~11 GB, import ~3 menit.
#    cu128 WAJIB untuk RTX 50xx (PC lab, RTX 5060 Ti) dan tetap jalan di RTX 3050.
#    Cukup ke node-2 karena hanya node-2 yang punya GPU.
#    Disk C sempit? Hapus dulu image lama di node-2 sebelum import (ruangnya dipakai ulang):

  --build-arg TORCH_INDEX_URL=https://download.pytorch.org/whl/cu128 \

# ------------------------------------------------------------
# 5. Konfigurasi runner
# ------------------------------------------------------------
kubectl create namespace testbed-jobs --dry-run=client -o yaml | kubectl apply -f -
export K8S_NAMESPACE=testbed-jobs
export IMAGE_PULL_POLICY=IfNotPresent
# Mode GPU: container job meminta nvidia.com/gpu=1 (otomatis jalan di node-2),
# sidecar membaca nvidia-smi -> kolom gpu_util_pct, gpu_mem_util_pct,
# gpu_mem_used_mb_device terisi.
export JOB_IMAGE=local/data-generator-job:gpu
export K8S_REQUEST_GPU=true
# Mode CPU (tanpa GPU): export JOB_IMAGE=local/data-generator-job:latest K8S_REQUEST_GPU=false
# Prometheus di-query oleh RUNNER di host (bukan dari Pod) -> pakai URL ingress
export PROMETHEUS_URL=http://prometheus.127.0.0.1.sslip.io
export PROM_RATE_WINDOW=1m        # jendela rate() untuk CPU/IO/network
export PROM_SCRAPE_LAG_SEC=25     # jeda sebelum query agar scrape terakhir masuk

# 6. Uji cepat (~25 menit): 1 sesi per skenario, durasi tetap 60 s

#     data GPU hanya dari nvidia-smi di sidecar.

#   --sessions 400 --misuse-frac 0.25 --duration-scale 0.5 -> ~29 jam (misuse 100, ~10 per skenario)

OUT=data-v2-$(date +%Y%m%d)

# Output per sesi:
#   <session>.csv             telemetri psutil dari sidecar
#   <session>.prometheus.csv  query_range Prometheus (step = --interval-sec)
#   <session>.merged.csv      gabungan berdasarkan timestamp (dipakai dataset.csv)
#   plan.json                 rencana sesi + parameter yang diundi (untuk --resume)
#   policy.json               ambang kebijakan sandbox (idle, batas sesi, memori GPU node)
#
# Kolom label & konteks (lihat README data_generator, "Skenario dan taksonomi CATS"):
#   label (benign/misuse), scenario_code (BENIGN/R1../P5/S1), label_class (-/K1/K2/K3),
#   user_id, session_age_sec, last_activity_age_sec, gpu_alloc_sec, quota_mem_mb,
#   prom_queue_length, prom_queue_wait_max_sec, prom_user_active_sessions,
#   param_* (parameter yang diundi = ground truth, JANGAN dipakai sebagai fitur model)
# Antrean (queue) hanya terisi di mode GPU: runner membuat Pod "waiter-*" yang
# Pending karena GPU dipegang sesi. Sesi paralel R6 = Pod "session-*-sib-N".
# Jika run dihentikan paksa, bersihkan sisa Pod: kubectl delete pod -n testbed-jobs --all
#
# Cek cepat sinyal per skenario setelah run:
python - <<'EOF'
import glob, pandas as pd
d = sorted(glob.glob("data-v2-*"))[-1]
df = pd.read_csv(f"{d}/dataset.csv")
cols = ["gpu_util_pct", "gpu_mem_used_mb_device", "cpu_pct", "last_activity_age_sec",
        "prom_queue_length", "prom_queue_wait_max_sec", "prom_user_active_sessions"]
print(df.groupby(["scenario_code", "job_type"])[cols].mean().round(1).to_string())
EOF

# ------------------------------------------------------------
# 8. Ambil metrik Prometheus secara manual (opsional)
# ------------------------------------------------------------
# Data Pod tetap ada di Prometheus setelah Pod dihapus (retensi default 10 hari),
# jadi sesi lama bisa di-query ulang. Nama Pod = session-<session_id dengan _ jadi ->.
P=http://prometheus.127.0.0.1.sslip.io
POD=session-misuse-idle-hoarding-0002

# Nilai sesaat
curl -sG $P/api/v1/query --data-urlencode \
  "query=sum(container_memory_working_set_bytes{namespace=\"testbed-jobs\",pod=\"$POD\",container=\"job\"})/1024/1024"

# Rentang waktu (time series), step 5 detik, 1 jam terakhir
curl -sG $P/api/v1/query_range \
  --data-urlencode "query=100*sum(rate(container_cpu_usage_seconds_total{namespace=\"testbed-jobs\",pod=\"$POD\",container=\"job\"}[1m]))" \
  --data-urlencode "start=$(date -d '-1 hour' +%s)" \
  --data-urlencode "end=$(date +%s)" \
  --data-urlencode "step=5"

# Daftar Pod sesi yang pernah tercatat
curl -sG $P/api/v1/label/pod/values --data-urlencode 'match[]={namespace="testbed-jobs"}'

# Query yang dipakai runner ada di PROMETHEUS_QUERIES (k8s_job_runner.py).
# Catatan kolom:
#   - CPU/memory/IO/restart/request/limit: container "job" saja (sidecar tidak ikut).
#   - network: per Pod. Skenario saat ini tidak memakai jaringan -> nilainya ~0.
#   - prom_cpu/mem request/limit selalu kosong (Pod tidak memasang requests/limits).
#   - io lokal (io_read_mb) = KUMULATIF MB; prom_io_read_mb = LAJU MB/s.
#   - prom_cpu_pct = rata-rata 1 menit -> tertinggal dibanding cpu_pct psutil.
